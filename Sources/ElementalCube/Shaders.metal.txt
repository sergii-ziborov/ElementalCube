#include <metal_stdlib>
using namespace metal;
struct Vertex { float4 position; float4 normal; float4 color; };
struct Uniforms { float4x4 projection; float4x4 model; float4 animation; float4 camera; float4 support; float4 world; };
struct Varying { float4 position [[position]]; float4 color; float3 normal; float3 world; float3 local; float2 fiber; };
vertex Varying scene_vertex(const device Vertex *v [[buffer(0)]], constant Uniforms &u [[buffer(1)]], uint id [[vertex_id]]) {
    Vertex a=v[id]; float3 p=a.position.xyz;
    if(u.world.y>0.5 && u.world.y<1.5) p+=a.normal.xyz*sin(u.animation.x*3+p.y*8+p.x*5)*.008;
    else if(u.world.y>1.5 && u.world.y<2.5) p+=a.normal.xyz*sin(u.animation.x*5+p.y*9-p.x*6)*.012;
    float4 world=u.model*float4(p,1);
    Varying o; o.position=u.projection*world; o.world=world.xyz;
    o.normal=normalize((u.model*float4(a.normal.xyz,0)).xyz);
    o.color=a.color; o.local=p; o.fiber=float2(a.position.w,a.normal.w); return o;
}
float coatHash(float2 p) { return fract(sin(dot(p,float2(127.1,311.7)))*43758.5453); }
float coatNoise(float2 p) {
    float2 i=floor(p),f=fract(p);f=f*f*(3-2*f);
    return mix(mix(coatHash(i),coatHash(i+float2(1,0)),f.x),mix(coatHash(i+float2(0,1)),coatHash(i+1),f.x),f.y);
}
float3 plushColor(float3 base,float3 p,float3 n,float wet,float layer) {
    float light=saturate(dot(n,normalize(float3(-.65,1.4,.9))));
    float3 shadow=base*float3(.64,.76,.98);
    float3 color=mix(shadow,base,.5+.5*light);
    color*=.91+.1*saturate(p.y+.5)+layer*.08;
    float strokes=coatNoise(float2(p.x*170+p.y*34,p.y*34));
    color*=.965+strokes*.065;
    color=mix(color,color*float3(.62,.76,.9),wet*.64);
    // Broad, quiet highlights read as velvet when dry and satin when wet.
    float sheen=pow(saturate(dot(n,normalize(float3(-.35,.55,1)))),wet>.5 ? 20. : 5.);
    return color+float3(.075,.09,.1)*sheen*(.25+wet*.75);
}
fragment float4 scene_fragment(Varying v [[stage_in]], constant Uniforms &u [[buffer(1)]]) {
    if(v.fiber.y< -1.) return float4(v.color.rgb*(.95+.05*sin(u.animation.x*7+v.local.y*14)),v.color.a);
    if(u.world.y>.5 && u.camera.w>.5) {
        float3 n=normalize(v.normal),view=normalize(float3(-.22,.18,1));
        float diffuse=saturate(dot(n,normalize(float3(-.65,1.4,.9))));
        float rim=pow(1-abs(dot(n,view)),2.);
        if(u.world.y>4.5) {
            float cloud=.035*sin(v.local.x*12.+v.local.y*9.+u.animation.x*.7);
            return float4(v.color.rgb*(.78+.22*diffuse)+float3(.08,.10,.10)*rim+cloud,.96);
        }
        if(u.world.y<1.5) {
            float spec=pow(saturate(dot(n,normalize(float3(-.4,.65,1)))),38.);
            float3 blue=v.color.rgb*(.6+.32*diffuse)+float3(.14,.30,.32)*rim+float3(.7,.95,1)*spec*.7;
            return float4(blue,.94);
        }
        if(u.world.y<2.5) return float4(mix(v.color.rgb,float3(1,.94,.50),saturate(v.local.y+.5)*.45)+float3(.07,.015,0),1);
        if(u.world.y<3.5) return float4(v.color.rgb*(.7+.3*diffuse)+float3(.15,.2,.19)*rim,.19);
        float3 p=v.local;float2 uv=abs(n.z)>.6 ? p.xy : (abs(n.x)>.6 ? p.zy : p.xz);
        float grain=coatNoise(uv*31.)*.045+coatNoise(uv*12.)*.07;
        return float4(v.color.rgb*(.5+.5*diffuse)*(.94+grain),1);
    }
    if(v.fiber.y>.5) {
        if(u.support.x>.5 && v.world.y<u.support.y+u.support.z*(v.world.x-u.support.w)+.065) discard_fragment();
        float t=v.fiber.y-1,across=v.fiber.x;
        float aa=max(fwidth(across),.045);
        float edge=1-smoothstep(1-aa*1.4,1.,abs(across));
        float strands=.72+.28*cos(across*19.+sin(t*7.)*.7);
        float alpha=edge*strands*(1-smoothstep(.78,1.,t));
        if(alpha<.08) discard_fragment();
        float3 color=plushColor(v.color.rgb,float3(0,.25,0),normalize(v.normal),0,t);
        return float4(color,alpha);
    }
    if(u.camera.w>1.5) return float4(plushColor(v.color.rgb,v.local,normalize(v.normal),u.animation.y,0),1);
    float diffuse=max(0.,dot(normalize(v.normal),normalize(float3(-.65,1.4,.9))));
    float3 lit=v.color.rgb*(.42+.58*diffuse);
    if(u.camera.w>0.5) {
        lit=v.color.rgb*(.56+.44*diffuse);
        lit*=1-u.animation.y*.27;
        lit+=float3(.005,.025,.05)*u.animation.y;
        float sheen=pow(max(0.,dot(normalize(v.normal),normalize(float3(-.4,.4,1)))),12.);
        lit+=float3(.12,.18,.20)*sheen*u.animation.y;
    }
    return float4(lit,v.color.a);
}

// Short overlapping shells fill the spaces between simulated guide tufts.
// No image textures or per-pixel world-space noise: the coat stays on the body.
vertex Varying coat_vertex(const device Vertex *vertices [[buffer(0)]],constant Uniforms &u [[buffer(1)]],
                           constant float4 &coat [[buffer(2)]],uint id [[vertex_id]]) {
    Vertex a=vertices[id];float3 p=a.position.xyz,n=a.normal.xyz;float layer=coat.x,wet=u.animation.y;
    float face=smoothstep(.25,.43,p.z)*smoothstep(.5,.9,n.z);
    float eyes=length(float2((abs(p.x)-.21)/.158,(p.y-.095)/.205));
    float mouth=length(float2(p.x/.115,(p.y+.17)/.12));
    float mask=mix(1.,smoothstep(.85,1.16,min(eyes,mouth)),face);
    float length=mix(.085,.025,wet)*mix(1.,.55,face)*mask;
    float3 groom=float3(.25,-.62,.04);groom-=n*dot(groom,n);
    float3 bend=groom*layer*layer*length*(.5+wet);
    bend.x+=clamp(coat.y,-3.,3.)*.008*layer*layer*(1-wet*.65);
    float4 world=u.model*float4(p+n*length*layer+bend,1);
    Varying o;o.position=u.projection*world;o.world=world.xyz;o.local=p;
    o.normal=normalize((u.model*float4(n,0)).xyz);o.color=a.color;o.fiber=float2(layer,mask);
    return o;
}
fragment float4 coat_fragment(Varying v [[stage_in]],constant Uniforms &u [[buffer(1)]]) {
    if(u.support.x>.5 && v.world.y<u.support.y+u.support.z*(v.world.x-u.support.w)+.065) discard_fragment();
    float layer=v.fiber.x,wet=u.animation.y;
    float3 n=normalize(v.normal);float3 p=v.local;
    float3 an=abs(p);float2 uv;
    if(an.z>=max(an.x,an.y)) uv=p.xy;
    else if(an.y>an.x) uv=p.xz;
    else uv=p.zy;
    float density=mix(66.,38.,wet);
    float2 coord=uv*density;
    float2 cell=floor(coord),f=fract(coord);float distance=4.;
    for(int y=-1;y<=1;y++) for(int x=-1;x<=1;x++) {
        float2 delta=float2(x,y),key=cell+delta;
        float2 root=.18+.64*float2(coatHash(key),coatHash(key+37.));
        distance=min(distance,length((f-delta-root)*float2(1.,mix(.75,.4,wet))));
    }
    float aa=clamp(length(fwidth(coord))*.45,.035,.55);
    float radius=mix(.66,.14,pow(layer,.8));
    float coverage=(1-smoothstep(radius-aa,radius+aa,distance))*(1-smoothstep(.72,1.,layer))*v.fiber.y;
    if(coverage<.08) discard_fragment();
    float3 color=plushColor(v.color.rgb,p,n,wet,layer);
    return float4(color,coverage);
}
struct Screen { float4 position [[position]]; float2 uv; };
vertex Screen background_vertex(uint id [[vertex_id]]) {
    float2 p=float2(id==1 ? 3 : -1,id==2 ? 3 : -1);
    return {float4(p, .999, 1),p*.5+.5};
}
float oval(float2 uv,float2 center,float2 radius) { return 1-smoothstep(.6,1.0,length((uv-center)/radius)); }
fragment float4 background_fragment(Screen v [[stage_in]], constant Uniforms &u [[buffer(1)]]) {
    float2 uv=v.uv; float x=uv.x+u.camera.x*.005;
    float3 sky=mix(float3(.90,.91,.75),float3(.53,.76,.79),smoothstep(.2,1.,uv.y));
    float aspect=u.camera.y/u.camera.z;
    float sun=oval(uv,float2(.72,.76),float2(.075,.075*aspect));
    float glow=exp(-length((uv-float2(.72,.76))*float2(1, .7))*7);
    sky+=float3(.12,.08,-.02)*glow;
    sky=mix(sky,float3(1,.96,.72),sun*.9);
    for(int i=0;i<4;i++) {
        float cx=fmod(.1+float(i)*.29+u.animation.x*.0007,1.3)-.1;
        float c=oval(uv,float2(cx,.76-float(i%2)*.13),float2(.13,.016*aspect));
        c=max(c,oval(uv,float2(cx+.05,.78-float(i%2)*.13),float2(.07,.022*aspect)));
        sky=mix(sky,float3(.96,.97,.87),c*.65);
    }
    float hill=.48+.055*sin(x*7)+.035*sin(x*18+1.4);
    sky=mix(sky,float3(.49,.68,.65),smoothstep(hill+.008,hill-.008,uv.y));
    float hill2=.41+.043*sin(x*9+2)+.03*sin(x*23);
    sky=mix(sky,float3(.33,.56,.53),smoothstep(hill2+.006,hill2-.006,uv.y));
    for(int i=0;i<32;i++) {
        float tx=float(i)/31.; float height=.048+fract(sin(float(i)*123.6)*456.7)*.06;
        float base=.39+.012*sin(tx*12);
        float tree=step(abs(x-tx),max(0.,(base+height-uv.y)*.24))*step(base,uv.y)*step(uv.y,base+height);
        sky=mix(sky,float3(.32,.52,.49),tree*.6);
    }
    sky=mix(sky,float3(.25,.43,.40),smoothstep(.35,.29,uv.y));
    float grain=fract(sin(dot(v.position.xy,float2(12.9898,78.233)))*43758.5453);
    if(u.world.x>5.5) sky=mix(sky,float3(.90,.74,.49),.22);
    else if(u.world.x>4.5) sky=mix(sky,float3(.65,.81,.88),.36);
    else if(u.world.x>3.5) sky=mix(sky,float3(.19,.29,.38),.52);
    else if(u.world.x>2.5) sky=mix(sky,float3(.53,.70,.43),.3);
    else if(u.world.x>1.5) sky=mix(sky,float3(sky.r*1.13,sky.g*.78,sky.b*.67),.75);
    else if(u.world.x>.5) sky=mix(sky,float3(sky.r*.82,sky.g*.94,min(1.,sky.b*1.13)),.8);
    return float4(sky+(grain-.5)*.012,1);
}
